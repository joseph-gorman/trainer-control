async function connectTrainer() { log("Requesting Bluetooth device with FTMS (0x1826)..."); try { // 1. Bluefy compatibility: Use full 128-bit UUID strings, acceptAllDevices fallback, // and explicitly declare optionalServices. trainerDevice = await navigator.bluetooth.requestDevice({ filters: [ { services: ['00001826-0000-1000-8000-00805f9b34fb'] }, { services: [0x1826] } ], optionalServices: [ '00001826-0000-1000-8000-00805f9b34fb', 0x1826, '00001818-0000-1000-8000-00805f9b34fb', // Cycling Power Service fallback 0x1818 ] }).catch(async (err) => { // Fallback for Bluefy when service filtering fails on iOS log(`Direct filter failed (${err?.message || err}). Trying open device picker...`, 'warn'); return await navigator.bluetooth.requestDevice({ acceptAllDevices: true, optionalServices: [ '00001826-0000-1000-8000-00805f9b34fb', 0x1826, '00001818-0000-1000-8000-00805f9b34fb', 0x1818 ] }); }); if (!trainerDevice) { throw new Error("No device selected."); } log(`Selected: ${trainerDevice.name || 'Unnamed Trainer'}. Connecting to GATT Server...`); trainerDevice.addEventListener('gattserverdisconnected', onTrainerDisconnected); const server = await trainerDevice.gatt.connect(); log("GATT connected. Discovering FTMS Primary Service..."); // Try full 128-bit UUID first, fallback to 16-bit let ftmsService; try { ftmsService = await server.getPrimaryService('00001826-0000-1000-8000-00805f9b34fb'); } catch { ftmsService = await server.getPrimaryService(0x1826); } // Acquire Control Point (0x2AD9) try { controlPointChar = await ftmsService.getCharacteristic('00002ad9-0000-1000-8000-00805f9b34fb'); } catch { controlPointChar = await ftmsService.getCharacteristic(0x2AD9); } // Step A: Request Control (OpCode 0x00) log("Requesting Trainer Control (OpCode 0x00)..."); await controlPointChar.writeValueWithResponse(new Uint8Array([0x00])); log("Trainer control acquired successfully!", "success"); // Acquire Indoor Bike Data notifications (0x2AD2) for live watts/cadence try { let indoorBikeDataChar; try { indoorBikeDataChar = await ftmsService.getCharacteristic('00002ad2-0000-1000-8000-00805f9b34fb'); } catch { indoorBikeDataChar = await ftmsService.getCharacteristic(0x2AD2); } await indoorBikeDataChar.startNotifications(); indoorBikeDataChar.addEventListener('characteristicvaluechanged', handleIndoorBikeData); log("Subscribed to live Indoor Bike Data (Watts/RPM).", "success"); } catch (e) { log(`Indoor Bike Data telemetry unavailable: ${e.message}`, "warn"); } updateTrainerUIConnected(true); // Initialize baseline resistance (30%) await setTrainerResistance(currentResistancePct); } catch (err) { const errorMsg = err?.message || (typeof err === 'object' ? JSON.stringify(err) : String(err)); log(`Trainer connection failed: ${errorMsg}`, "err"); updateTrainerUIConnected(false); } }
